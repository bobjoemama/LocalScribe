cmake_minimum_required(VERSION 3.24)
project(localscribe_canary LANGUAGES C CXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  message(FATAL_ERROR "LocalScribe Canary requires Apple Silicon")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_REAL_MODEL_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_ENABLE_VALIDATION_HOOKS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_METAL ON CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/pin.json" pin)
string(JSON revision GET "${pin}" revision)
string(JSON archive_sha256 GET "${pin}" archiveSha256)
include(FetchContent)
FetchContent_Declare(transcribe
  URL "https://codeload.github.com/handy-computer/transcribe.cpp/tar.gz/${revision}"
  URL_HASH "SHA256=${archive_sha256}"
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(transcribe)
# Upstream uses CMAKE_SOURCE_DIR for its public headers; add the correct
# subproject include directory without patching the pinned upstream source.
target_include_directories(transcribe PUBLIC "${transcribe_SOURCE_DIR}/include")
add_library(localscribe-canary SHARED bridge.cpp)
target_link_libraries(localscribe-canary PRIVATE transcribe)
target_compile_definitions(localscribe-canary PRIVATE LOCALSCRIBE_CANARY_REVISION="${revision}")
target_compile_options(localscribe-canary PRIVATE -Wall -Wextra -Werror)
set_target_properties(localscribe-canary PROPERTIES
  BUILD_WITH_INSTALL_NAME_DIR TRUE INSTALL_NAME_DIR "@rpath")
add_executable(canary-contract-test contract-test.cpp)
target_link_libraries(canary-contract-test PRIVATE localscribe-canary)
target_compile_options(canary-contract-test PRIVATE -Wall -Wextra -Werror)
